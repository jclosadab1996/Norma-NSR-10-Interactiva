export const ej1 = (() => {
  const V = 30, h = 12, L = 20, B = 40, theta = 15, Kzt = 1, Kd = 0.85;
  const I = factorImportanciaViento('II');
  const Kh = kz('B', h);
  const qh = presionVelocidad({ V, Kz: Kh, Kzt, Kd, I });
  const kz6 = kz('B', 6);
  const qz6 = presionVelocidad({ V, Kz: kz6, Kzt, Kd, I });
  const res = presionesEdificioRigido({ qh, qzBarlovento: qh, cerramiento: 'cerrado', L, B, h, theta, direccion: 'normal' });
  const f = (id) => res.filas.find((x) => x.clave === id);
  const pb6 = G_RIGIDA * 0.8 * qz6;
  const pbh = G_RIGIDA * 0.8 * qh;
  const qp = presionVelocidad({ V, Kz: kz('B', 13), Kzt, Kd, I });
  return {
    V, h, L, B, theta, I, Kh, qh, kz6, qz6, pb6, pbh, hl: h / L, lb: L / B, res, qp,
    bar: f('barlovento'), sot: f('sotavento'), lat: f('lateral'), cb: f('cubierta-barlovento'), cs: f('cubierta-sotavento'),
    cpb: cpCubiertaBarlovento(h / L, theta), cpSot: cpMuroSotavento(L / B),
    pParBar: presionParapeto(qp, 'barlovento'), pParSot: presionParapeto(qp, 'sotavento'),
    pAlero: presionAleroBarlovento(qh),
  };
})();

export const ej2 = (() => {
  const { qh, qz6, h, L, B, theta } = ej1;
  const res = presionesEdificioRigido({ qh, qzBarlovento: qh, qiPositiva: qz6, cerramiento: 'parcial', L, B, h, theta, direccion: 'normal' });
  const f = (id) => res.filas.find((x) => x.clave === id);
  const resCons = presionesEdificioRigido({ qh, qzBarlovento: qh, cerramiento: 'parcial', L, B, h, theta, direccion: 'normal' });
  const g = (id) => resCons.filas.find((x) => x.clave === id);
  const ri = reduccionRi(40, L * B * h);
  return { res, bar: f('barlovento'), cb: f('cubierta-barlovento'), cs: f('cubierta-sotavento'), sot: f('sotavento'), consCb: g('cubierta-barlovento'), ri };
})();

export const ej3 = (() => {
  const V = 30, h = 6, L = 20, B = 30, theta = 20, Kd = 0.85;
  const I = factorImportanciaViento('I');
  const Kh = kz('C', h, 1);
  const qh = presionVelocidad({ V, Kz: Kh, Kd, I });
  const gci = 0.18;
  const zonas = ['1', '2', '2E', '3'].map((z) => {
    const g = gcpf(z, theta);
    return { z, g, pMas: presionBajaAltura({ qh, gcpf: g, gcpi: gci }), pMenos: presionBajaAltura({ qh, gcpf: g, gcpi: -gci }) };
  });
  const a = anchoZonaA(Math.min(L, B), h);
  const area = 2;
  const gm = gcpMuro(area, '5', theta);
  const pPos = presionComponentes({ qh, gcp: gm.positivo, gcpi: -gci }); // presión externa + presión interna negativa
  const pNeg = presionComponentes({ qh, gcp: gm.negativo, gcpi: gci }); // succión externa + presión interna positiva
  const hl = h / L;
  return { V, h, L, B, theta, I, Kh, qh, zonas, a, area, gm, pPos, pNeg, hl };
})();

## Idea central del capítulo

El Método 2 de B.6.5 (procedimiento analítico) ya te dio la **presión por velocidad** $q_z$ (ecuación B.6.5-13, vista en B.6a/B.6b) y el factor de efecto de ráfaga $G$. Este tramo del capítulo responde la pregunta práctica que sigue: **¿qué presión neta actúa sobre cada superficie del edificio?** La respuesta se arma con tres ingredientes:

1. **La clasificación del cerramiento** (cerrado, parcialmente cerrado o abierto) → coeficiente de presión interna $GC_{pi}$ (Figura B.6.5-2).
2. **El coeficiente de presión externa** $C_p$ (o $GC_p$, $GC_{pf}$, $C_N$, $C_f$ según el caso) que depende de la forma, de la relación $h/L$, de $L/B$ y del ángulo de cubierta $\theta$ (Figuras B.6.5-3 a B.6.5-19).
3. **La ecuación de combinación** de B.6.5.12 a B.6.5.15, que multiplica y suma lo anterior con el signo correcto.

<Callout tipo="importante" titulo="Convención de signos y combinación">
Una presión **positiva** actúa **hacia** la superficie y una **negativa** actúa **hacia afuera** (<NormaRef s="B.6.5.12.1.1" pag="B-33" />). Las presiones internas y externas **se combinan algebraicamente** para hallar el caso más crítico (<NormaRef s="B.6.5.12.1.2" pag="B-33" />): por eso casi siempre hay que evaluar **dos** condiciones, una con $+GC_{pi}$ y otra con $-GC_{pi}$.
</Callout>

Dos conceptos de estructura recorren todo el texto:

- **SPRFV** (Sistema Principal de Resistencia a Fuerzas de Viento): pórticos, muros de cortante, diafragmas… que reciben la carga **global** del edificio. Se calcula con $C_p$ y se aplica sobre áreas grandes.
- **Componentes y revestimientos**: fachadas, vidrios, correas, paneles de cubierta, anclajes… que reciben presiones **locales** más intensas. Se calculan con $GC_p$, que ya incluye el efecto de ráfaga y depende del **área efectiva** de viento.

## Del capítulo anterior: presión por velocidad $q_z$

Para situar el resto, recuerda que la presión por velocidad a la altura $z$ se calcula con la ecuación B.6.5-13 (<NormaRef s="B.6.5.10" pag="B-32" />): $q_z = 0.613\,K_z K_{zt} K_d V^2 I$ en N/m² con $V$ en m/s, y que $q_h$ es el mismo valor evaluado a la altura media de la cubierta $h$. El coeficiente 0.613 se usa siempre que no haya registros climáticos suficientes para justificar otro valor. También recuerda la regla de clasificaciones múltiples (<NormaRef s="B.6.5.9.4" pag="B-31" />): si un edificio cumple a la vez las definiciones de "abierto" y "parcialmente cerrado", se clasifica como **abierto**; si no cumple ninguna de las dos, es **cerrado**.

## B.6.5.11 — Coeficientes de presión y fuerza

### B.6.5.11.1 — Coeficiente de presión interna

Los coeficientes $GC_{pi}$ se leen de la Figura B.6.5-2 según la clasificación del cerramiento de B.6.5.9 (<NormaRef s="B.6.5.11.1" pag="B-32" />):

<TablaNorma numero="Figura B.6.5-2" titulo="Coeficientes de presión interna GCpi" pag="B-49" nota="Signo positivo: presión sobre las superficies internas; negativo: succión. Los valores se usan con qz o qh según B.6.5.12 (nota 2). | Clasificación del cerramiento | $GC_{pi}$ |
|---|---|
| Edificios abiertos | 0.00 |
| Edificios parcialmente cerrados | +0.55 y −0.55 |
| Edificios cerrados | +0.18 y −0.18 |

</TablaNorma>

Un edificio **parcialmente cerrado** es el que tiene una gran abertura en la pared a barlovento (por ejemplo, una puerta rota o abierta durante la tormenta): la presión interna se "infla" y casi triplica la de un edificio cerrado. De ahí que la clasificación pese tanto en el diseño. Las condiciones formales (<NormaRef s="B.6.2" pag="B-22" />) son $A_o > 1.10\,A_{oi}$, $A_o > 0.37$ m² o $A_o > 0.01\,A_g$ (el menor) y $A_{oi}/A_{gi} \le 0.20$; el edificio es **abierto** si $A_o \ge 0.8\,A_g$ en cada pared.

<Figura src="/img/titulo-b/figura-b6-5-2-gcpi.png" alt="Tabla de la Figura B.6.5-2 con los coeficientes de presión interna según la clasificación del cerramiento" numero="Figura B.6.5-2" pie="Coeficientes de presión interna GCpi (PDF)." pag="B-49" />

#### B.6.5.11.1.1 — Factor de reducción $R_i$ para edificios de gran volumen

Si el edificio es **parcialmente cerrado** y tiene **un solo espacio sin particiones** (una bodega, un coliseo), el aire interior tiene tanto volumen que la presión interna no alcanza a desarrollarse completamente: $GC_{pi}$ se multiplica por $R_i \le 1.0$ (<NormaRef s="B.6.5.11.1.1" pag="B-32" />).

<Ecuacion num="B.6.5-14" titulo="Factor de reducción Ri por gran volumen">

$$
R_i = 0.5\left(1+\frac{1}{\sqrt{1+\dfrac{V_i}{11000\,A_{og}}}}\right)\le 1.0
$$

- $A_{og}$: área total de aberturas en el cerramiento (paredes y cubierta), en m².
- $V_i$: volumen interno sin particiones, en m³.

</Ecuacion>

$R_i$ vale 1.0 cuando el volumen es pequeño frente a las aberturas y tiende a 0.5 cuando es muy grande. En el edificio del Ejemplo 2 ($A_{og}$ = 40 m², $V_i$ = {(ej1.L * ej1.B * ej1.h).toFixed(0)} m³) $R_i$ = {ej2.ri.toFixed(3)}: casi no reduce.

<Callout tipo="atencion" titulo="Constante 11000 impresa en el PDF">
El PDF imprime el coeficiente **11000** en el denominador. La fórmula equivalente de ASCE 7 en unidades SI usa un número menor (≈ 6950); no se corrige aquí porque el texto de la NSR-10 es claro, pero conviene que el diseñador lo tenga presente: con 11000 la reducción es algo menos fuerte que con 6950.
</Callout>

### B.6.5.11.2 — Coeficientes de presión externa

**SPRFV** (<NormaRef s="B.6.5.11.2.1" pag="B-32" />): los $C_p$ salen de las Figuras B.6.5-3 (muros y cubiertas), B.6.5-4 (domos) y B.6.5-5 (cubiertas en arco). Para edificios bajos se dan combinados con el factor de ráfaga en la Figura B.6.5-7 ($GC_{pf}$).

**Componentes y revestimientos** (<NormaRef s="B.6.5.11.2.2" pag="B-32" />): los $GC_p$ salen de las Figuras B.6.5-8A a B.6.5-14.

<Callout tipo="importante" titulo="No separes G de Cp en los coeficientes combinados">
Los productos $GC_{pf}$ (Fig. B.6.5-7) y $GC_p$ (Figs. B.6.5-8A a B.6.5-14) **ya incluyen** el efecto de ráfaga. No les apliques $G$ de nuevo ni los descompongas en $G$ y $C_p$ (<NormaRef s="B.6.5.11.2.1" pag="B-32" />, <NormaRef s="B.6.5.9" pag="B-31" />).
</Callout>

## B.6.5.12 — Fuerzas de viento de diseño en edificios cerrados o parcialmente cerrados

### B.6.5.12.1 — General

Tres reglas generales (<NormaRef s="B.6.5.12.1" pag="B-33" />):

1. **Signos:** positiva = hacia la superficie; negativa = hacia afuera.
2. **Carga crítica:** las presiones internas y externas se combinan algebraicamente.
3. **Áreas aferentes mayores de 65 m²:** los elementos de revestimiento y componentes con área aferente mayor de 65 m² se pueden diseñar con las reglas del SPRFV.

### B.6.5.12.2 — Sistemas principales de resistencia a fuerzas de viento

#### B.6.5.12.2.1 — Edificios rígidos de cualquier altura

<Ecuacion num="B.6.5-15" titulo="Presión de diseño del SPRFV, edificios rígidos">

$$
p = q\,G\,C_p - q_i\,(GC_{pi})\qquad \text{en N/m}^2
$$

- $q = q_z$ para **muros a barlovento**, evaluada a la altura $z$ sobre el terreno.
- $q = q_h$ para **muros a sotavento, muros laterales y cubiertas**, evaluada a la altura $h$.
- $q_i = q_h$ para **todas** las superficies de edificios **cerrados** y para presiones internas **negativas** en parcialmente cerrados.
- $q_i = q_z$ para presiones internas **positivas** en parcialmente cerrados, con $z$ = nivel de la **abertura más alta** que podría afectar la presión interna positiva (puede tomarse conservadoramente $q_i = q_h$).
- $G$: factor de efecto de ráfaga (B.6.5.8; 0.85 para rígidos).
- $C_p$: Figuras B.6.5-3 o B.6.5-5. $GC_{pi}$: Figura B.6.5-2.

</Ecuacion>

La exposición para $q$ y $q_i$ es la definida en B.6.5.6.3, y las presiones se aplican **simultáneamente** en muros a barlovento y sotavento y en cubiertas (<NormaRef s="B.6.5.12.2.1" pag="B-33" />). Si el sitio puede tener **arrastre de fragmentos** por el viento, los vidrios de los 20 m inferiores que no sean resistentes al impacto ni estén protegidos se tratan como **aberturas** (<NormaRef s="B.6.5.9.3" pag="B-31" />), lo que puede convertir un edificio cerrado en parcialmente cerrado.

#### B.6.5.12.2.4 — Parapetos

Los parapetos de edificios rígidos, bajos o flexibles con cubiertas planas, a dos aguas o a cuatro aguas aumentan las cargas del SPRFV con una presión neta combinada sobre el parapeto:

$$
p_p = q_p\,GC_{pn}
$$

- $p_p$: presión neta combinada de las caras anterior y posterior del parapeto (positiva hacia el frente del parapeto, negativa hacia afuera). El PDF imprime la unidad "N/m", pero es una **presión** en N/m².

- $GC_{pn}$ = **+1.5** para el parapeto a barlovento y **−1.0** para el parapeto a sotavento.

</Ecuacion>

</TablaNorma>

<Ecuacion num="B.6.5-19" titulo="Excentricidad e de estructuras flexibles">

$$
e=\frac{e_Q+1.7\,I_{\bar z}\sqrt{(g_Q Q e_Q)^2+(g_R R e_R)^2}}{1+1.7\,I_{\bar z}\sqrt{(g_Q Q)^2+(g_R R)^2}}
$$

- $e_Q$: excentricidad $e$ de estructuras rígidas según la Figura B.6.5-6.
- $e_R$: distancia entre el centro de cortante elástico y el centro de masa de cada piso.
- $I_{\bar z}$, $g_Q$, $Q$, $g_R$ y $R$: definidos en B.6.5.8.

</Ecuacion>

#### B.6.5.12.4.1 — Edificios bajos y edificios con $h \le 18.0$ m

### B.6.5.15.1 — Estructuras y equipos sobre cubiertas en edificios con $h \le 18.0$ m

La fuerza sobre estructuras y equipos de cubierta con $A_f$ **menor de $0.1\,B\,h$** en edificios con $h \le 18.0$ m se calcula con la ecuación de fuerza (el PDF la cita como "Ec. B.6-38"; corresponde a B.6.5-26) **multiplicada por 1.9**. Ese factor puede reducirse linealmente de 1.9 a 1.0 a medida que $A_f$ aumenta de $0.1\,B\,h$ a $B\,h$ (<NormaRef s="B.6.5.15.1" pag="B-37" />). Para un equipo de 15 m² en un edificio de $B$ = 40 m y $h$ = 12 m ($0.1Bh$ = 48 m², $Bh$ = 480 m²) el factor es 1.9, porque 15 m² es menor que 48 m². Con $A_f$ = 100 m² valdría {factorEquipoCubierta(100, 40, 12).toFixed(2)}.

## B.6.6 — Método 3: procedimiento de túnel de viento

El Método 3 reemplaza los coeficientes de las figuras por **mediciones sobre un modelo a escala** del edificio y su entorno, sometido a un flujo de aire que reproduce la capa límite atmosférica. Es la herramienta para formas, entornos o respuestas fuera del alcance de los Métodos 1 y 2.

### B.6.6.1 — Alcance

Los ensayos de túnel de viento **deben** usarse donde lo exija B.6.5.2: edificios o estructuras que no cumplan las condiciones de regularidad de B.6.5.1, de forma irregular o con características de respuesta inusuales y que no se diseñen con literatura reconocida (<NormaRef s="B.6.6.1" pag="B-37" />). Además, el ensayo **se permite en lugar de los Métodos 1 y 2 para cualquier edificio o estructura**.

### B.6.6.2 — Condiciones de ensayo

Los ensayos para determinar variaciones y promedios de fuerzas y presiones con fluidos distintos del aire deben cumplir las siguientes condiciones (<NormaRef s="B.6.6.2" pag="B-38" />):

<TablaNorma numero="B.6.6.2" titulo="Condiciones que debe cumplir el ensayo de túnel de viento" pag="B-38">

| Numeral | Condición |
|---|---|
| B.6.6.2.1 | La capa límite de la atmósfera natural se modela considerando la variación de la velocidad del viento con la altura |
| B.6.6.2.2 | Las escalas de macro y microlongitud de la turbulencia longitudinal se modelan aproximadamente a la misma escala que el modelo del edificio |
| B.6.6.2.3 | El edificio modelado y los alrededores (estructuras y topografía) son geométricamente similares a los reales; en edificios bajos (B.6.5.1) se permite escalar una sola categoría de exposición |
| B.6.6.2.4 | El área proyectada del modelo y sus alrededores es menor de **8 %** del área de la sección de ensayo, salvo que se corrija por bloqueo |
| B.6.6.2.5 | Se considera el gradiente de presión longitudinal en la sección de ensayo |
| B.6.6.2.6 | Se minimizan los efectos del número de Reynolds sobre presiones y fuerzas |
| B.6.6.2.7 | La respuesta de la instrumentación es consistente con las mediciones requeridas |

</TablaNorma>

### B.6.6.3 — Respuesta dinámica

Los ensayos para determinar la **respuesta dinámica** (aceleraciones, desplazamientos) deben cumplir B.6.6.2, y el modelo estructural y su análisis deben reflejar la **distribución de masa, la rigidez y el amortiguamiento** (<NormaRef s="B.6.6.3" pag="B-38" />).

### B.6.6.4 — Limitaciones

La **variación de la velocidad básica con la dirección** no se permite en el análisis de túnel de viento, a menos que el análisis de las velocidades de viento cumpla B.6.5.4.2 (uso de datos climáticos regionales con análisis estadístico aprobado) (<NormaRef s="B.6.6.4.1" pag="B-38" />). En otras palabras: sin esa justificación se debe suponer el viento básico igual en todas las direcciones.

<Callout tipo="practica" titulo="Cuándo pensar en túnel de viento">
Torres esbeltas y flexibles, formas curvas o muy irregulares, edificios rodeados de otros altos que canalizan el viento, cubiertas colgantes o ligeras de gran luz y estadios son candidatos típicos. El ensayo no es gratuito, así que en proyectos comunes se usa el Método 1 o 2; en los complejos, el ensayo suele reducir presiones respecto a las de las figuras, pero siempre bajo las limitaciones de B.6.6.2 y B.6.6.4.
</Callout>

## Errores comunes

<Callout tipo="error" titulo="Qué suele salir mal">
- Aplicar **solo una** condición de $GC_{pi}$: hay que evaluar $+GC_{pi}$ y $-GC_{pi}$ y quedarse con la más crítica para cada elemento.
- Usar $q_h$ en el muro a barlovento en vez de $q_z$ (o al revés para sotavento, laterales y cubierta): la presión en barlovento **varía con la altura**.
- Multiplicar $GC_{pf}$ o $GC_p$ por $G$: esos coeficientes **ya incluyen** el efecto de ráfaga.
- Olvidar la **segunda fila** de $C_p$ de la cubierta a barlovento: puede ser **presión** sobre la cubierta y gobernar el diseño de la correa o del anclaje.
- Clasificar como cerrado un edificio con grandes ventanales sin protección en zona de arrastre de fragmentos: esos vidrios cuentan como aberturas (<NormaRef s="B.6.5.9.3" pag="B-31" />).
- Diseñar componentes con áreas efectivas grandes (> 65 m²) sin aprovechar que pueden diseñarse como SPRFV, o usar $C_p$ del SPRFV en un clip de fachada: son sistemas distintos.
- Olvidar los casos de carga torsionales y los de viento oblicuo (Figura B.6.5-6) en el SPRFV de edificios no exentos.
- Usar las tablas de edificios cerrados para una cubierta libre (edificio abierto): ahí rigen los $C_N$.
</Callout>

## Resumen en una línea

**Clasifica el cerramiento ($GC_{pi}$), lee $C_p$ (o $GC_{pf}$, $GC_p$, $C_N$, $C_f$) en la figura que corresponda y combina $q\,G\,C_p - q_i\,GC_{pi}$ con ambos signos de presión interna; si el edificio es irregular o inusual, recurre al túnel de viento.**
